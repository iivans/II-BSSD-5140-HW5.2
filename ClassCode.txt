# Dynamic Programming implementation of LCS problem with autocorrect functionality
def lcs(X, Y): 
    # find the length of the strings 
    m = len(X) 
    n = len(Y) 
    
    # declaring the array for storing the dp values 
    L = [[None]*(n + 1) for i in range(m + 1)] 

    # Following steps build L[m+1][n+1] in bottom up fashion 
    for i in range(m + 1): 
        for j in range(n + 1): 
            if i == 0 or j == 0: 
                L[i][j] = 0
            elif X[i-1] == Y[j-1]: 
                L[i][j] = L[i-1][j-1] + 1
            else: 
                L[i][j] = max(L[i-1][j], L[i][j-1]) 
 
    return L[m][n] 

# Autocorrect function using LCS
def autocorrect(word, pool):
    results = {w: lcs(word, w) for w in pool}
    
    # Sort by LCS length in descending order
    results = {k: v for k, v in sorted(results.items(), key=lambda item: item[1], reverse=True)}
    
    # Get the LCS of assumed best match
    commonality = list(results.items())[0][1]
    
    # Check if LCS is more than half of entered word
    if commonality > len(word) // 2:
        print(f"I think you meant: {list(results.keys())[0]}")
    else:
        print("I have no idea what you meant.")

# A Naive recursive Python program to find minimum number
# of operations to convert s1 to s2
def editDistDP(s1, s2, m, n):
    # If first string is empty, the only option is to
    # insert all characters of second string into first
    if m == 0:
        return n

    # If second string is empty, the only option is to
    # remove all characters of first string
    if n == 0:
        return m

    # If last characters of two strings are same, nothing
    # much to do. Get the count for
    # remaining strings.
    if s1[m - 1] == s2[n - 1]:
        return editDistDP(s1, s2, m - 1, n - 1)

    # If last characters are not same, consider all three
    # operations on last character of first string,
    # recursively compute minimum cost for all three
    # operations and take minimum of three values.
    return 1 + min(editDistDP(s1, s2, m, n - 1),    # Insert
                   editDistDP(s1, s2, m - 1, n),    # Remove
                   editDistDP(s1, s2, m - 1, n - 1) # Replace
                   )

def main():
    pool = ['goose', 'cat', 'danger', 'panic']
    
    # Input from user
    word = input("Please enter a word from the list:\n{0}.\n".format(pool))
    word = word.lower()  # Convert to lowercase so capitalizing isn't a step
    
    # Calculate edit distance for each word in pool
    results = {w: editDistDP(word, w, len(word), len(w)) for w in pool}
    
    # Sort results based on the fewest edit distance
    results = {k: v for k, v in sorted(results.items(), key=lambda item: item[1])}
    
    # Get the edit distance of assumed answer
    commonality = list(results.items())[0][1]
    
    # Confidence check: If common letters < half the length, reject
    if commonality < len(word) / 2:
        print(f"I think you meant", list(results.keys())[0])
    else:
        print("I have no idea what you meant.")

if __name__ == "__main__":
    main()
